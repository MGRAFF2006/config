# Does NOT replace KDE (laptop) or DWM (desktop). Install alongside, pick session in SDDM / startx.
#
# Install:  sudo pacman -S --needed - < packages/hyprland-dms.txt
# Remove:   ~/Documents/config/scripts/remove-hyprland-dms.sh
#
# Start DMS from Hyprland autostart only (exec-once / hl.on) — do NOT
# `systemctl --user enable dms` while Plasma remains a daily session,
